% --- Slide 9: Paradoja de Russell ---
\begin{frame}{El Colapso: La Paradoja de Russell (1901)}
    \begin{center}
        Sea $R$ el conjunto de todos los conjuntos que no se contienen a sí mismos:
        
        \Large
        \begin{equation*}
            R = \{x \mid x \notin x\}
        \end{equation*}
        
        \vspace{0.8cm}
        \alert{\textbf{¿$R \in R$?}}
    \end{center}
\end{frame}

% --- Slide 10: ZF y la Reconstrucción ---
\begin{frame}{La Reconstrucción: Zermelo-Fraenkel (ZF)}
    \begin{center}
        La destrucción de la Comprensión Irrestricta y el nacimiento del \textbf{Axioma de Separación (Especificación)}:
        
        \Large
        \vspace{0.5cm}
        \begin{equation*}
            \forall A \, \exists Y \, \forall x \, (x \in Y \iff x \in A \land P(x))
        \end{equation*}
        
        \normalsize
        \vspace{0.8cm}
    \end{center}
\end{frame}
