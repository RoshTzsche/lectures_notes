\begin{frame}{Teoría de Conjuntos: El Lenguaje}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            Definimos las operaciones que gobiernan la lógica de datos:
            \begin{itemize}
                \item \textbf{Unión ($\cup$):} Fusión de mundos.
                \item \textbf{Intersección ($\cap$):} Consistencia compartida.
                \item \textbf{Diferencia ($\setminus$):} Exclusión lógica.
                \item \textbf{Potencia $\mathcal{P}(A)$:} El conjunto de todos los subconjuntos.
            \end{itemize}
            \begin{block}{Teorema Crucial}
                $|\mathcal{P}(A)| = 2^{|A|}$ \\
                \footnotesize La explosión combinatoria nace aquí.

% --- Slide 4: Paradoja de Russell ---
\begin{frame}{El Colapso: La Paradoja de Russell}
    \begin{columns}
        \begin{column}{0.55\textwidth}
            ¿Puede un conjunto contenerse a sí mismo?
            Sea $R$ el conjunto de todos los conjuntos que \alert{no} se contienen a sí mismos:
            \begin{equation*}
                R = \{x \mid x \notin x\}
            \end{equation*}
            \textbf{La Contradicción:}
            \begin{itemize}
                \item Si $R \in R \implies R \notin R$ (Absurdo)
                \item Si $R \notin R \implies R \in R$ (Absurdo)
            \end{itemize}
            \textit{Lección: Incluso la lógica pura tiene grietas si no se define con rigor.}
        \end{column}
        \begin{column}{0.45\textwidth}
            \centering
            \includegraphics[width=0.9\textwidth]{http://googleusercontent.com/image_collection/image_retrieval/12052078414567897431}

        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Equipolencia: Contar sin Números}
    \begin{itemize}
        \item Dos conjuntos tienen el mismo tamaño (cardinalidad) si existe una \textbf{biyección} entre ellos.
        \item No necesitamos contar hasta el infinito; necesitamos emparejar elementos.
    \end{itemize}

% --- Slide 6: El Infinito ---
\begin{frame}{El Infinito no es un Número}
    Existen diferentes "tamaños" de infinito. 
    \begin{itemize}
        \item \textbf{Conjuntos Contables ($|\mathbb{N}|$):} 
        Podemos listarlos: $\mathbb{Z}$ (enteros) y $\mathbb{Q}$ (racionales) son \alert{del mismo tamaño} que $\mathbb{N}$.
        \item \textbf{El Continuo ($|\mathbb{R}|$):} 
        ¿Es posible listar todos los números reales entre 0 y 1?
    \end{itemize}
    \begin{exampleblock}{Sorpresa Matemática}
        $|\mathbb{N}| = |\mathbb{Z}| = |\mathbb{Q}|$ \\
        A pesar de que $\mathbb{N} \subset \mathbb{Z} \subset \mathbb{Q}$, hay una biyección entre ellos.
    \end{exampleblock}
\end{frame}

% --- Slide 7: Argumento Diagonal ---
\begin{frame}{La Gran Ruptura: Argumento Diagonal}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            Georg Cantor (1891) demostró que $|\mathbb{R}| > |\mathbb{N}|$.
            \vspace{0.2cm}
            \begin{enumerate}
                \item Supongamos que tenemos una lista de todos los reales.
                \item Construimos un número $d$ cuya $n$-ésima cifra difiere de la $n$-ésima cifra del $n$-ésimo número de la lista.
                \item $d$ \alert{no puede estar} en la lista.
            \end{enumerate}
            \textbf{Conclusión:} El infinito de los reales es mayor.
        \end{column}
        \begin{column}{0.5\textwidth}
            \centering
            \includegraphics[width=\textwidth]{http://googleusercontent.com/image_collection/image_retrieval/8753770717026798672}
            \\ \small La matriz de la diagonalización
        \end{column}
    \end{columns}
\end{frame}
