% --- Slide 7: El Lenguaje y Sintaxis ---
\begin{frame}{2.1 El Lenguaje como Sistema Formal}
    \textbf{Sintaxis del Cálculo Proposicional}
    
    \vspace{0.4cm}
    \begin{itemize}
        \item \textbf{Alfabeto} $\Sigma$ y el conjunto infinito de todas las cadenas posibles $\Sigma^*$.
        \item \textbf{Variables lógicas:} $p, q, r, \dots$
        \item \textbf{Conectivos:} $\neg, \land, \lor, \to, \leftrightarrow$
    \end{itemize}

    \vspace{0.6cm}
    \alert{\textbf{Fórmula Bien Formada (FBF) por Recursión:}}
    \begin{enumerate}
        \item Toda variable proposicional es una FBF.
        \item Si $\varphi$ es una FBF, entonces $(\neg\varphi)$ es una FBF.
        \item Si $\varphi$ y $\psi$ son FBFs, entonces $(\varphi * \psi)$ es una FBF (donde $*$ es un conectivo binario).
    \end{enumerate}
\end{frame}

% --- Slide 8: Semántica ---
\begin{frame}{2.2 Semántica: El Mundo del Significado}
    \begin{center}
        \Large Asignando "Verdad" a marcas vacías.
        
        \vspace{0.4cm}
        \begin{equation*}
            v: \text{VAR} \to \{0, 1\}
        \end{equation*}
        \normalsize
    \end{center}
    
    \vspace{0.4cm}
    \begin{itemize}
        \item \textbf{Tautología:} Verdadera bajo toda valuación posible ($v(\varphi) = 1$).
        \item \textbf{Contradicción:} Falsa bajo toda valuación posible ($v(\varphi) = 0$).
        \item \textbf{Consecuencia Semántica ($\Gamma \models \varphi$):} Si todas las fórmulas en $\Gamma$ son verdaderas, entonces obligatoriamente $\varphi$ es verdadera.
    \end{itemize}
\end{frame}

% --- Slide 9: Teoría de la Demostración ---
\begin{frame}{2.3 Teoría de la Demostración}
    \begin{columns}
        \begin{column}{0.4\textwidth}
            \textbf{El Motor Mecánico}
            \vspace{0.5cm}
            
            \large
            Modus Ponens:
            \begin{equation*}
                \frac{\varphi \quad \varphi \to \psi}{\psi}
            \end{equation*}
        \end{column}
        \begin{column}{0.6\textwidth}
            \textbf{Derivación Formal ($\Gamma \vdash \varphi$)}
            
            \vspace{0.3cm}
            \small La prueba concebida no como un acto de iluminación humana, sino como una \textbf{transformación puramente mecánica de símbolos} basada en reglas previas.
            
            \vspace{0.6cm}
            \alert{\textbf{Teorema de Corrección (Soundness):}}
            \begin{equation*}
                \Gamma \vdash \varphi \implies \Gamma \models \varphi
            \end{equation*}
            \footnotesize \textit{Si una fórmula se puede probar sintácticamente por la máquina, entonces es semánticamente verdadera en el universo.}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 10: El Clímax (NAND) ---
\begin{frame}{El Clímax: Completitud Funcional}
    \begin{center}
        \Large
        ¿Es posible expresar la totalidad del universo proposicional con una sola pieza?
        
        \vspace{0.5cm}
        \alert{\textbf{El Operador NAND ($\uparrow$)}}
    \end{center}
    
    \vspace{0.5cm}
    \textbf{Reto Algebraico:} Construye el universo abstracto.
    \begin{itemize}
        \item Expresa $\neg p$ usando únicamente $p \uparrow q$
        \item Expresa $p \land q$ usando únicamente $\uparrow$
        \item Expresa $p \lor q$ usando únicamente $\uparrow$
    \end{itemize}
    
    \vspace{0.5cm}
    \begin{center}
        \textit{Con un solo bloque, fundamos las bases del cómputo moderno.}
    \end{center}
\end{frame}
