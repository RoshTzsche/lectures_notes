% INTRODUCCIÓN: EL LOGICISMO
% ==========================================

\begin{frame}{El Logicismo: Principia Mathematica}
    \begin{definicion}[Logicismo]
        Programa fundacional que postula la reducción estricta de la matemática a la lógica. Se articula en dos tesis:
        \begin{enumerate}
            \item Toda constante matemática es definible mediante vocabulario puramente lógico.
            \item Todo teorema matemático es derivable a partir de axiomas lógicos mediante reglas de inferencia formales.
        \end{enumerate}
    \end{definicion}
    
    \vspace{0.4cm}
    \alert{\textbf{El Isomorfismo de la Consistencia:}} \\
    La reducción buscaba garantizar que si el sistema lógico $\mathcal{L}$ es consistente ($\mathcal{L} \not\vdash \bot$), la aritmética heredaría automáticamente dicha consistencia.
\end{frame}

% ==========================================
% LA ARITMÉTICA DE PEANO
% ==========================================

\begin{frame}{Sintaxis: La Firma de Peano ($\mathcal{L}_{PA}$)}
    \begin{definicion}[Firma Aritmética]
        El lenguaje de primer orden $\mathcal{L}_{PA} = (\mathcal{F}, \mathcal{P}, ar)$ consta de:
    \end{definicion}
    
    \vspace{0.2cm}
    \begin{columns}
        \begin{column}{0.5\textwidth}
            \alert{\textbf{Símbolos de Función ($\mathcal{F}$):}}
            \begin{itemize}
                \item $0$ (Constante, $ar=0$)
                \item $S$ (Sucesor, $ar=1$)
                \item $+, \cdot$ (Operadores, $ar=2$)
            \end{itemize}
        \end{column}
        \begin{column}{0.5\textwidth}
            \alert{\textbf{Símbolos de Predicado ($\mathcal{P}$):}}
            \begin{itemize}
                \item $=$ (Igualdad, $ar=2$)
                \item $<$ (Orden, $ar=2$)
            \end{itemize}
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{Los Axiomas de Peano ($PA$)}
    \footnotesize
    \textbf{1. Cero y Sucesor:} 
    $$PA_1: \forall x \, (S(x) \neq 0)$$
    $$PA_2: \forall x \forall y \, (S(x) = S(y) \to x = y)$$
    
    \textbf{2. Definición Recursiva de Operaciones:}
    $$PA_3: \forall x \, (x + 0 = x) \quad \quad PA_4: \forall x \forall y \, (x + S(y) = S(x + y))$$
    $$PA_5: \forall x \, (x \cdot 0 = 0) \quad \quad PA_6: \forall x \forall y \, (x \cdot S(y) = (x \cdot y) + x)$$
    
    \textbf{3. Esquema Axiomático de Inducción:}
    $$PA_7: (\phi(0) \land \forall x \, (\phi(x) \to \phi(S(x)))) \to \forall x \, \phi(x)$$
\end{frame}

% ==========================================

\begin{frame}{Propiedades Metamatemáticas}
    \begin{definicion}[Consistencia Sintáctica]
        Un sistema formal $F$ es consistente si no existe una fórmula $\phi$ tal que $F \vdash \phi$ y $F \vdash \neg \phi$. Equivalentemente, $F \not\vdash \bot$.
    \end{definicion}
    
    \vspace{0.4cm}
    \begin{definicion}[Enumerabilidad Recursiva]
        Un sistema $F$ es recursivamente enumerable si el conjunto de los números de Gödel de sus teoremas $T = \{ \ulcorner \phi \urcorner \mid F \vdash \phi \}$ es el rango de una función recursiva parcial. Existe una máquina de Turing que lista todos y cada uno de los teoremas de $F$.
    \end{definicion}
\end{frame}

% ==========================================
% LIMITACIONES SEMÁNTICAS (TARSKI)
% ==========================================

\begin{frame}{El Límite Semántico: El Teorema de Tarski}
    \textbf{La Paradoja del Mentiroso:} \textit{"Este enunciado es falso"}. \\
    \vspace{0.2cm}
    Carente de valor formal, pues la "verdad" es un concepto semántico, no sintáctico.
    
    \vspace{0.4cm}
    \begin{teorema}[Indefinibilidad de la Verdad de Tarski, 1936]
        Para cualquier sistema formal $F$ que contenga a la aritmética de Peano, no existe una fórmula $\text{True}(x)$ en el lenguaje de $F$ tal que para toda fórmula $\phi$:
        $$F \vdash \text{True}(\ulcorner \phi \urcorner) \leftrightarrow \phi$$
    \end{teorema}
    
    \alert{Conclusión:} Ningún sistema lo suficientemente potente puede definir su propio predicado de verdad.
\end{frame}

% ARITMETIZACIÓN DE LA SINTAXIS
% ==========================================

\begin{frame}{Aritmetización: La Función de Gödel}
    \begin{definicion}[Número de Gödel]
        Isomorfismo inyectivo entre el álgebra sintáctica y $\mathbb{N}$. Sea una asignación $c: \Sigma \to \mathbb{N}$. Para una secuencia de símbolos $\phi = s_1 s_2 \dots s_k$, su número de Gödel es:
        $$\ulcorner \phi \urcorner = \prod_{i=1}^k p_i^{c(s_i)}$$
        donde $p_i$ es el $i$-ésimo número primo.
    \end{definicion}
    
    \vspace{0.2cm}
    \begin{definicion}[Predicado de Demostrabilidad]
        $\text{Prov}_F(x)$ es una relación recursiva primitiva tal que $\text{Prov}_F(\ulcorner \phi \urcorner)$ es verdadera en $\mathbb{N}$ si y solo si $F \vdash \phi$.
    \end{definicion}
\end{frame}

% ==========================================
% TEOREMAS DE INCOMPLETITUD
% ==========================================

\begin{frame}{El Punto Fijo Sintáctico}
    \begin{lema}[Lema Diagonal / Lema de Autorreferencia]
        Sea $F$ un sistema formal que extiende $PA$. Para toda fórmula $\psi(x)$ con una variable libre, existe un enunciado $\delta$ tal que:
        $$F \vdash \delta \leftrightarrow \psi(\ulcorner \delta \urcorner)$$
    \end{lema}

    \vspace{0.4cm}
    \alert{\textbf{Construcción de la Oración $G$:}} \\
    Sea $\psi(x) := \neg \text{Prov}_F(x)$. Por el Lema Diagonal, existe $G$ tal que:
    $$F \vdash G \leftrightarrow \neg \text{Prov}_F(\ulcorner G \urcorner)$$
\end{frame}

\begin{frame}{El Primer Teorema de Gödel}
    \begin{columns}
        \begin{column}{0.3\textwidth}
            \begin{center}
                \includegraphics[height=0.55\textheight]{../../assets/Kurt-Godel.jpg} \\
                \vspace{0.2cm}
                \textbf{Kurt Gödel} \\
                \small (1906 -- 1978)
            \end{center}
        \end{column}
        \begin{column}{0.7\textwidth}
            \begin{teorema}[Primer Teorema de Incompletitud, 1931]
                Cualquier sistema axiomático $F$ recursivamente enumerable, que sea consistente y lo suficientemente expresivo para contener la aritmética ($F \supseteq PA$), es incompleto.
            \end{teorema}
            
            \vspace{0.2cm}
            \textbf{Corolario:} $\exists G$ tal que $F \not\vdash G$ y $F \not\vdash \neg G$. \\
            Puesto que $G$ no es demostrable, y $G$ afirma no serlo, $G$ es \textbf{verdadera en $\mathbb{N}$}, separando irremediablemente verdad de demostrabilidad ($\models \neq \vdash$).
        \end{column}
    \end{columns}
\end{frame}

\begin{frame}{El Segundo Teorema de Gödel}
    \begin{teorema}[Segundo Teorema de Incompletitud]
        Sea $F$ un sistema axiomático consistente que contenga a la aritmética ($F \supseteq PA$). Entonces, $F$ no puede demostrar su propia consistencia.
    \end{teorema}
    
    \vspace{0.6cm}
    \textbf{Formulación Lógica:} \\
    Definimos $\text{Con}(F) \equiv \neg \text{Prov}_F(\ulcorner \bot \urcorner)$. El teorema establece que:
    $$F \not\vdash \text{Con}(F)$$
\end{frame}

\begin{frame}{Indecidibilidad Computacional: Alan Turing}
    \begin{columns}
        \begin{column}{0.3\textwidth}
            \begin{center}
                \includegraphics[height=0.55\textheight]{../../assets/Alan-Turing.jpg} \\
                \vspace{0.2cm}
                \textbf{Alan Turing} \\
                \small (1912 -- 1954)
            \end{center}
        \end{column}
        \begin{column}{0.7\textwidth}
            \begin{definicion}[Máquina de Turing]
                Un autómata definido por la 7-tupla $M = \langle Q, \Gamma, b, \Sigma, \delta, q_0, F \rangle$, donde $\delta: (Q \setminus F) \times \Gamma \to Q \times \Gamma \times \{L, R\}$ es la función de transición.
            \end{definicion}
            
            \begin{teorema}[Indecidibilidad del Halting Problem, 1936]
                El lenguaje universal $HALT = \{ \langle M, w \rangle \mid M \text{ se detiene con entrada } w \}$ es recursivamente enumerable, pero \textbf{no es decidible} (no es recursivo). No existe máquina de Turing que compute la función característica de $HALT$.
            \end{teorema}
        \end{column}
    \end{columns}
\end{frame}

% ==========================================
% DIAGONALIZACIÓN UNIVERSAL
% ==========================================

\begin{frame}{La Topología del Límite Absoluto: Diagonalización}
    \begin{teorema}[Teorema de Cantor]
        Para todo conjunto $A$, la cardinalidad de su conjunto potencia es estrictamente mayor que la de $A$. Es decir, $\not\exists f: A \to \mathcal{P}(A)$ que sea suprayectiva.
    \end{teorema}
    
    \vspace{0.4cm}
    \alert{\textbf{El Conjunto Diagonal (El Punto Ciego):}} \\
    Para cualquier función evaluadora $f$, el elemento que colapsa el sistema se construye sistemáticamente como:
    $$S = \{ x \in A \mid x \notin f(x) \}$$
    \vspace{0.2cm}
    \footnotesize \textit{Cantor lo usó en los reales, Gödel en la sintaxis aritmética, Turing en la computabilidad. Las reglas de un sistema absoluto son su propia arma de destrucción.}
\end{frame}
